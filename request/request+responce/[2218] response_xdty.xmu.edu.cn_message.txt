<script>
    // td dom渲染后更新td
    let tdTimer = null;

    function isIOSNotchScreen() {
        // 检测是否为 iOS 设备
        const isIOS = /iPad|iPhone|iPod/.test(navigator.userAgent);
        if (!isIOS) return false;

        // 检测是否存在刘海屏的安全区域
        const screenWidth = window.innerWidth;
        const screenHeight = window.innerHeight;
        const isNotch = isIOS && screenWidth >= 375 && screenHeight >= 812;
        return isNotch;
    }

    document.addEventListener("DOMContentLoaded", function () {
        if (isIOSNotchScreen()) {
            document.body.classList.add("fix");
        }
    });
    var vm = new Vue({
        el: "#content",
        data() {
            return {
                dataList: {},
                viewIndex: 0,
                stadium_id: 0,
                category_id: 0,
                project_name: '',
                can_choose: 0,
                is_app: 0,
                is_done: false,
                is_academy: 1,
                isModalVisible: false,
                persionList: [],
                persion: null,
                student_num: null,
                inputerror: false,
                radio: 1,
                add_auth: null,
                personal_num: 0,
                team_num: 0,
                btnText: '请先选择场次',
                dis: 'dis',
                venue_appointment_min_num: 0,// 是否开启最少预约人数限制0关闭大于0开启
                confirm_time: '半小时',
                is_can_choose: 1,
                is_can_choose_tips: '',
                originTimeSlotList: [],
                selectedSlots: []
            }
        },
        computed: {
            max() {
                const {personal_num = "", team_num = ""} = this.dataList;
                return this.is_academy == 1 ? personal_num : team_num;
            },
            limit() {
                let temp =
                    this.radio == 1
                        ? this.dataList.personal_num
                        : this.dataList.team_num;
                return (temp - 1) | 0;
            },
            // 当前选中日期对应的时段列表（核心联动）
            currentTimeSlots() {
                if (!this.dataList.date_list || !this.originTimeSlotList.length) return []
                const currentDate = this.dataList.date_list[this.viewIndex]?.date_int
                return this.originTimeSlotList.filter(item => item.date === currentDate)
            }
        },
        methods: {
            getClass(slot) {
                const {personal_num = 0, team_num = 0} = this.dataList;
                const max = parseInt(this.is_academy) === 1 ? personal_num : team_num;
                if (slot.selected >= max) {
                    return 'unavailable'
                }
                return slot.status
            },
            changeDate(index) {
                this.viewIndex = index
                this.$nextTick(() => {
                    this.restoreSelectedStatus();
                });
                this.setBtnStatus()
            },
            restoreSelectedStatus() {
                document.querySelectorAll('.slot-cell').forEach(el => {
                    const date = el.dataset.date;
                    const column_id = el.dataset.areaId;
                    const interval_id = el.dataset.intervalId;

                    if (!date || !column_id || !interval_id) return;

                    const key = `${date}_${column_id}_${interval_id}`;
                    const isSelected = this.selectedSlots.some(item => item.key === key);
                    if (isSelected) {
                        $(el).addClass('bg-2');
                    } else {
                        $(el).removeClass('bg-2');
                    }
                });
            },
            deal(i) {
                this.persionList.splice(i, 1)
                this.setBtnStatus()
            },
            add: debounce(function () {
                if (!this.persion || !this.stadium_id) {
                    layer.msg("请输入姓名或学号");
                    return;
                }
                const areaIds = [];
                this.selectedSlots.forEach(item => {
                    areaIds.push(item.item.area_id);
                });
                let that = this;
                layer.load(2, {shade: false});
                request("/index/stadium/check_add_auth", {
                    stadium_id: this.stadium_id,
                    name: this.persion,
                    student_num: this.student_num,
                    area_ids: areaIds.join(","),
                    venue_id: this.venue_id,
                    category_id: this.category_id
                })
                    .then((rs) => {
                        layer.closeAll();
                        const {uid = null} = rs.data;
                        if (rs.status == 0) {
                            layer.msg(rs.info);
                        } else if (rs.status == 1) {
                            // 重复时跳过更新为新版
                            that.persionList = that.persionList.filter((item) => {
                                return item.id != uid;
                            });
                            that.persionList = [
                                ...that.persionList,
                                {
                                    name: that.persion,
                                    value: that.student_num,
                                    id: uid,
                                },
                            ];
                            layer.msg("添加成功");
                            that.setBtnStatus()
                            that.hideModal();
                        } else if (rs.status == -1) {
                            layuiLoginOpen();
                        }
                    })
                    .catch(() => {
                        layer.closeAll();
                        that.hideModal();
                    });
            }, 200),
            showModal() {
                if (this.is_can_choose === 0) {
                    layer.msg(this.is_can_choose_tips)
                    return
                }
                if (this.persionList.length >= this.limit) {
                    layer.msg("已选场次名额不足，请更换场次");
                    return;
                }
                this.persion = null;
                this.student_num = null;
                this.isModalVisible = true;
            },
            hideModal() {
                this.isModalVisible = false;
            },
            set_list: function (v) {
                this.dataList = v;
                this.personal_num = v.personal_num;
                this.team_num = v.team_num;
                this.originTimeSlotList = JSON.parse(JSON.stringify(v.time_slot_list))
            },
            changeView: function (index) {
                if (index < this.viewIndex) {
                    if (index >= 1) {
                        this.scrollToPosition(index - 1);
                    }
                } else {
                    if (index + 2 <= this.dataList.interval.length) {
                        this.scrollToPosition(index + 1);
                    }
                }

                this.viewIndex = index;
            },
            scrollToPosition(index) {
                document
                    .getElementsByClassName("m-tab-item")
                    [index].scrollIntoView({
                    behavior: "smooth",
                });
            },
            setBtnStatus() {
                const selectedCount = this.selectedSlots.length;
                if (this.venue_appointment_min_num > 0) {
                    //开启了最少预约人数限制
                    const yet = selectedCount > 0 ? 1 : 0;
                    if (yet + this.persionList.length < this.venue_appointment_min_num) {
                        if (yet > 0) {
                            this.btnText = (this.persionList.length > 0 ? '还需添加 ' : '请最少添加 ') + (this.venue_appointment_min_num - 1 - this.persionList.length) + ' 名随行人员'
                        } else {
                            this.btnText = '请先选择场次'
                        }
                        this.dis = 'dis';
                    } else {
                        this.btnText = '确认预约'
                        this.dis = '';
                    }
                } else {
                    this.btnText = selectedCount >= 1 ? '确认预约' : '请先选择场次'
                    this.dis = selectedCount >= 1 ? '' : 'dis';
                }
            },
            toConfirm() {
                if (this.is_can_choose === 0) {
                    layer.msg(this.is_can_choose_tips)
                    return
                }
                let price = 0
                let list = this.selectedSlots.map(i => {
                    price += parseFloat(i.item.price)
                    return i.item
                });
                price = price.toFixed(2)
                if (this.venue_appointment_min_num > 0) {
                    if (this.persionList.length + 1 < this.venue_appointment_min_num) {
                        const btnText = (this.persionList.length > 0 ? '还需添加 ' : '请最少添加 ') + (this.venue_appointment_min_num - 1 - this.persionList.length) + ' 名随行人员'
                        layer.msg(btnText)
                        return false
                    }
                }
                if (list.length === 0) {
                    layer.msg("请选择场次");
                    return false;
                }
                const self = this
                loadingPlugin.showLoading();
                request('/index/stadium/chooseVerify', {
                    stadium_id: getQueryVariable('stadium_id'),
                    selected: JSON.stringify(list),
                    venue_id: getQueryVariable('venue_id'),
                    is_academy: self.is_academy,
                    ids: self.radio === 1 ? self.persionList.map(item => item.id).join(',') : ''
                }, 2).then(rs => {
                    if (rs.status) {
                        const is_academy = this.is_academy;
                        const academy_name = '';
                        const mark = '';
                        list = JSON.stringify(list)
                        const stadium_id = getQueryVariable('stadium_id');
                        const venue_id = getQueryVariable('venue_id');
                        const category_id = getQueryVariable('category_id')
                        const project_name = this.project_name;
                        const name = this.name;
                        const p_count = self.persionList.length;
                        const params = {
                            stadium_id,
                            venue_id,
                            category_id,
                            project_name,
                            list,
                            name,
                            is_academy,
                            academy_name,
                            mark,
                            price
                        };
                        if (this.radio === 1) {
                            params.p_count = p_count;
                            params.ids = self.persionList.map(item => item.id).join(',');
                        }else{
                            params.p_count = 0
                            params.ids = ''
                        }
                        params.price = price > 0 ? (params.p_count + 1) * price : 0
                        this.navigateToPage('./confirm.html', params)
                    }
                }).catch(() => {
                }).finally(() => {
                    loadingPlugin.hideLoading();
                })
            },
            init() {
                vm.is_app = parseInt($.cookie("COOKIE_IS_APP_UI"));

                var venue_id = this.venue_id;
                var stadium_id = this.stadium_id;
                var user_range = this.user_range;

                request("/stadium/interval/getInterval", {
                    venue_id: venue_id,
                    stadium_id: stadium_id,
                    user_range: user_range,
                    category_id: this.category_id
                }).then((rs) => {
                    if (rs.status == 0) {
                        layer.msg(rs.info);
                    } else if (rs.status == 1) {
                        vm.set_list(rs.data);
                        vm.add_auth = rs.data.add_auth;
                        vm.venue_appointment_min_num = rs.data.venue_appointment_min_num
                        vm.confirm_time = rs.data.confirm_time
                        vm.neighbor = rs.data.neighbor
                        vm.is_can_choose = rs.data.is_can_choose
                        vm.is_can_choose_tips = rs.data.is_can_choose_tips
                        vm.is_done = true;
                        this.$nextTick(function () {
                            layui.use(['form'], function () {
                                const form = layui.form;
                                form.render()
                                form.on('radio(type)', function (data) {
                                    vm.is_academy = data.value
                                })
                            })
                        })
                    } else if (rs.status == -1) {
                        layuiLoginOpen();
                    }
                });
            },
            choose(e, item, cellEl) {
                const target = $(cellEl)
                if (item.status !== 'available' || target.hasClass('unavailable')) {
                    if (item.lock_reason) {
                        layer.msg(item.lock_reason)
                    }
                    return
                }
                if (this.is_can_choose === 0) {
                    layer.msg(this.is_can_choose_tips)
                    return
                }
                const key = `${item.date}_${item.column_id}_${item.interval_id}`;
                const isSelected = target.hasClass('bg-2');
                if (isSelected) {
                    // 取消选中：从数组移除
                    target.removeClass('bg-2');
                    this.selectedSlots = this.selectedSlots.filter(i => i.key !== key);
                } else {
                    // 选中：加入数组
                    target.addClass('bg-2');
                    this.selectedSlots.push({
                        key: key,
                        item: {
                            date: target.attr('data-date'),
                            week: target.attr('data-week'),
                            week_msg: target.attr('data-week-msg'),
                            area_name: target.attr('data-area-name'),
                            interval_time: target.attr('data-interval-time'),
                            interval_id: target.attr('data-interval-id'),
                            area_id: target.attr('data-area-id'),
                            price: target.attr('data-price')
                        }
                    })
                }
                this.setBtnStatus()
                return true
            },
            changeRadio(val) {
                if (val === this.radio) {
                    return false
                }
                this.is_academy = val;
                this.radio = val;
                $(".slot-cell").each(function (index, el) {
                    $(el).removeClass("bg-2")
                });
                this.selectedSlots = []
                this.can_choose = 0;
                this.setBtnStatus()
            },
            preview(e, img_url) {
                if (img_url == "") {
                    layer.msg("暂无预览图");
                    return false;
                }
                var auto_img = {};
                var img = new Image();
                img.src = img_url;
                img.onload = function () {
                    var max_height = $(window).height() - 100;
                    var max_width = $(window).width();
                    var rate1 = max_height / img.height;
                    var rate2 = max_width / img.width;
                    var rate3 = 1;
                    var rate = Math.min(rate1, rate2, rate3);
                    auto_img.height = img.height * rate;
                    auto_img.width = img.width * rate;
                    layer.open({
                        type: 1,
                        title: false,
                        area: ["300px", "300px"],
                        skin: "layui-layer-nobg", //没有背景色
                        shadeClose: true,

                        closeBtn: 1,
                        content:
                            "<img src='" +
                            img_url +
                            "' width=100% height='100%' class='u-radius'>",
                    });
                };
            },
            navigateToPage(path, params) {
                // 过滤掉值为 null/undefined/空字符串 的参数，避免无效参数
                const validParams = Object.entries(params).reduce((acc, [key, value]) => {
                    if (value != null) {
                        // 数组自动转逗号分隔字符串，其他类型直接转字符串
                        acc[key] = Array.isArray(value) ? value.join(',') : String(value);
                    }
                    return acc;
                }, {});
                const queryString = new URLSearchParams(validParams).toString();
                window.location.href = queryString ? `${path}?${queryString}` : path;
            }
        },
        updated: function () {
            layui.use(["layer", "form"], function () {
                const form = layui.form;
                form.render();
            });
        },
    });

    $(document).ready(function () {

        vm.project_name = getQueryVariable('project_name')
        vm.name = getQueryVariable('name')
        vm.category_id = getQueryVariable('category_id')
        vm.init()
    })
